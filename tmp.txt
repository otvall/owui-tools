import json
from typing import Any


def _build_tool_data_bridge(
    result: Any,
    title: str,
    tool_id: str,
) -> str:
    data_json = _safe_json_for_html(result)
    title_json = json.dumps(title, ensure_ascii=False)
    tool_id_json = json.dumps(tool_id, ensure_ascii=False)

    return (
        '<script id="iv-tool-data" type="application/json">'
        f"{data_json}"
        "</script>"

        "<script>(function(){"

        # -------------------------
        # getToolData
        # -------------------------
        "function read(id){"
            "var el=document.getElementById(id);"
            "if(!el)return null;"
            "try{return JSON.parse(el.textContent);}"
            "catch(e){return null;}"
        "}"

        "var data=read('iv-tool-data');"
        "window.getToolData=function(){return data;};"

        # -------------------------
        # metadata
        # -------------------------
        f"var title={title_json};"
        f"var toolId={tool_id_json};"

        # -------------------------
        # registry
        # -------------------------
        "try{"
            "var host=window.parent;"

            "if(host && host!==window){"

                # создаём registry один раз
                "if(!host.__OWUI_VIZ_REGISTRY__){"
                    "var items=new Map();"

                    "host.__OWUI_VIZ_REGISTRY__={"

                        "register:function(entry){"
                            "var key=entry.messageId+':'+entry.toolId;"
                            "items.set(key,entry);"
                        "},"

                        "list:function(){"
                            "return Array.from(items.values());"
                        "}"

                    "};"
                "}"

                # -------------------------
                # получаем messageId
                # -------------------------
                "var frame=window.frameElement;"

                "var msgEl="
                    "frame && "
                    "frame.closest && "
                    "frame.closest('[id^=\"message-\"]');"

                "var messageId=null;"

                "if(msgEl && msgEl.id){"
                    "messageId=msgEl.id.replace(/^message-/,'');"
                "}"

                # -------------------------
                # регистрация
                # -------------------------
                "if(messageId){"

                    "host.__OWUI_VIZ_REGISTRY__.register({"
                        "title:title,"
                        "messageId:messageId,"
                        "toolId:toolId"
                    "});"

                    "console.debug("
                        "'[viz-registry] registered',"
                        "{title:title,messageId:messageId,toolId:toolId}"
                    ");"

                "}else{"
                    "console.warn("
                        "'[viz-registry] messageId not found'"
                    ");"
                "}"

            "}"

        "}catch(e){"
            "console.error('[viz-registry]',e);"
        "}"

        "})();</script>"
    )
